\documentclass{standalone}
\usepackage{tikz}
\begin{document}
\definecolor{gridgray}{RGB}{213,213,213}
\definecolor{crimson}{RGB}{214,0,54}
\begin{tikzpicture}[color=black, >=stealth, scale=0.3, font=\scriptsize\bfseries]
  % Grid
  \draw[gridgray, thin] (-10,-10) grid (10,10);
  % Axes
  \draw[<->, thick] (-10.9,0) -- (10.9,0);
  \draw[<->, thick] (0,-10.9) -- (0,10.9);
  % Scale labels (even values only)
  \foreach \n in {-10,-8,-6,-4,-2,2,4,6,8,10} {
    \node[below, inner sep=2pt] at (\n,-0.15) {\n};
    \node[left, inner sep=2pt] at (-0.15,\n) {\n};
  }
  % Quartic:  y = 0.0076x^4 - 0.0069x^3 - 0.2777x^2 + 1.1484x + 0.7934
  \draw[<->, line width=1.2pt, crimson, smooth, samples=200, domain=-8.11:6.85]
    plot (\x, {0.0076*(\x)^4 - 0.0069*(\x)^3 - 0.2777*(\x)^2 + 1.1484*\x + 0.7934});
\end{tikzpicture}

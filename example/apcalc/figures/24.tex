\documentclass{standalone}
\usepackage{tikz}
\begin{document}
% TI-84 style calculator screen, 4:3 aspect ratio (8cm x 6cm).
% Window: x in [-5.5,7.5], y in [-5,5].
% Function: y = |(x+2)(x-4)|/4 - 4
\begin{tikzpicture}[x=0.61538cm, y=0.6cm, >=stealth]
  % screen border (exactly the viewing window -> 4:3)
  \draw[rounded corners=2pt, line width=1pt] (-5.5,-5) rectangle (7.5,5);

  \begin{scope}
    \clip (-5.5,-5) rectangle (7.5,5);

    % axes
    \draw[->] (-5.5,0) -- (7.5,0);
    \draw[->] (0,-5) -- (0,5);

    % minor tick marks (every 1 unit)
    \foreach \x in {-5,-3,-1,1,3,5,7}
      \draw ([yshift=-2.5pt]\x,0) -- ([yshift=2.5pt]\x,0);
    \foreach \y in {-5,-3,-1,1,3,5}
      \draw ([xshift=-2.5pt]0,\y) -- ([xshift=2.5pt]0,\y);

    % major tick marks (every 2 units)
    \foreach \x in {-4,-2,2,4,6}
      \draw ([yshift=-4pt]\x,0) -- ([yshift=4pt]\x,0);
    \foreach \y in {-4,-2,2,4}
      \draw ([xshift=-4pt]0,\y) -- ([xshift=4pt]0,\y);

    % curve: y = |(x+2)(x-4)|/4 - 4
    \draw[line width=1pt, smooth, samples=300, domain=-5.5:7.5]
      plot (\x, {abs((\x+2)*(\x-4))/4 - 4});

    % tick labels drawn over the curve (it passes through x = -4, 6, y = -2)
    \foreach \x in {-4,-2,2,4,6}
      \node[below=4pt, fill=white, inner sep=1pt] at (\x,0) {\scriptsize $\x$};
    \foreach \y in {-4,-2,2,4}
      \node[left=4pt, fill=white, inner sep=1pt] at (0,\y) {\scriptsize $\y$};
    \node[below left=1pt] at (0,0) {\scriptsize $0$};
  \end{scope}
\end{tikzpicture}
\end{document}
